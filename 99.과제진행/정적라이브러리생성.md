MKINSTALLDIRS=mkinstalldirs

BASE_ARCH = aan aca xcc oan xce xcg ocn
SIM_ARCH = sim
CLEAN_ARCH = $(BASE_ARCH) i686

ifdef INPUT_ARCH
BUILD_ARCH=$(shell for arch in $(BASE_ARCH);do for tmp in $(INPUT_ARCH); do echo $$tmp | grep -w $$arch; done;done)
else
BUILD_ARCH=$(BASE_ARCH)
endif

UT_PATH=Test/ut
include $(RULE_DIR)/unittest_rule.mk

all:
	for arch in $(BUILD_ARCH); do \
		$(MAKE) -f Makefile.$$arch $(MKFLAGS); \
	done

sim:
	for arch in $(SIM_ARCH); do \
		$(MAKE) -f Makefile.$$arch $(MKFLAGS); \
	done

clang:
	for arch in $(SIM_ARCH); do \
		$(MAKE) -f Makefile.$$arch cc=clang CC=clang++ $(MKFLAGS); \
	done

.DEFAULT:
	for arch in $(BUILD_ARCH); do \
		$(MAKE) -f Makefile.$$arch $@; \
	done

clean:
	for arch in $(CLEAN_ARCH); do \
		$(MAKE) -f Makefile.$$arch $@; \
	done

cleanall:
	for arch in $(CLEAN_ARCH); do \
		$(MAKE) -i -f Makefile.$$arch $@; \
	done

cleansim:
	rm -rf /OBJ/*/i686/vccb*
	rm -rf ../ITSCP_VCCB/OBJ/*/i686/itscp_vccb*
	rm -rf ../../../Middleware/OAM_LIB/DBI/OBJ/*
	rm -rf ../../../Middleware/OAM_LIB/SFTP/SFTP/OBJ/*
	rm -rf ../../../Middleware/OAM_LIB/VPMLIB/OBJ/*
	rm -rf ../../../Middleware/OAM_LIB/SWMGT/OBJ/*
	rm -rf ../../../Middleware/OAM_LIB/OAMLIB/FRAMEWORK/OBJ/*
	rm -rf ../../../Middleware/OAM_LIB/LIB_RULE_CHECK/RULECHECKER/dummy/OBJ/*
	rm -rf ../../../Middleware/OAM_LIB/OAMLIB/FMLIB/OBJ/
	rm -rf ../../../Application/NR_CALL_APP/Base/DR_LIB/OBJ/*	
	rm -rf ../../../Middleware/NR_CALL_LIB/IF/*/OBJ/*/
	rm -rf ../../../Middleware/MW_LIB/MFS/OBJ/*
	rm -rf ../../../3rdParty/CONFD_LIB/OBJ/*
	rm -rf ../../../Middleware/FEATURE_LIB/OBJ/*
	rm -rf ../../../Application/NR_CALL_APP/Tools/SIM/HAS_SIM/OBJ/*
	$(YANG_REMOVE_EXECUTE_COMMAND)	

asan:
	$(YANG_REMOVE_EXECUTE_COMMAND)
	@echo -e "\033[33m"MODE MEM TEST ... COMPILE ASAN"\033[0m"
	@make cleanall
	$(MAKE) -f Makefile.i686 -s SANITIZER_TYPE=ASAN $(MKFLAGS);

ubsan:
	$(YANG_REMOVE_EXECUTE_COMMAND)
	@echo -e "\033[33m"MODE MEM TEST ... COMPILE UBSAN"\033[0m"
	@make cleanall
	$(MAKE) -f Makefile.i686 -s SANITIZER_TYPE=UBSAN $(MKFLAGS);

with_itscp:
	for arch in $(BUILD_ARCH); do \
		$(MAKE) -f Makefile.$$arch; $(MKFLAGS) \
		$(MAKE) -f Makefile.$$arch -C "../ITSCP_VCCB"; $(MKFLAGS) \
	done

vclean:
	echo "This command is supported by Test/ut. Currently in the VCCB directory."

install: $(UT_BUILD)
	@$(MKINSTALLDIRS) $(PKG_DIR)/ne/config
	@$(MKINSTALLDIRS) $(PKG_DIR)/ne/bin/patch
	for arch in $(BUILD_ARCH); do \
		$(MAKE) -f Makefile.$$arch install; \
	done

doc:
	/usr/local/doxygen-1.8/bin/doxygen

cleandoc:
	rm -rf $(PWD)/DOXYGEN_DOC
