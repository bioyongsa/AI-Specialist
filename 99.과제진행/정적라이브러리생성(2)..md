MODE_IFLAGS =
ifeq ($(SIM_OSBIT),32)
MODE_LDFLAGS = -m32 -g

MODE_CXXFLAGS = -m32 -Wall -fPIC -g
else
MODE_LDFLAGS = -g
MODE_CFLAGS = -Wall -fPIC -g
MODE_CXXFLAGS = -Wall -fPIC -g
endif
ifneq ($(TOOLLIB),$(empty))
MODE_LDFLAGS += -L$(TOOLLIB) -Wl,-rpath,$(TOOLLIB)
endif
MODE_LDFLAGS += -lrt
