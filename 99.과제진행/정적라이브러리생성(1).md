ccFlagOverride=-O0 -fgnu89-inline -Wno-error
ldFlagOverride=-Wl,--warn-unresolved-symbols

$(STATIC_LIB): $(LOAD_LIBS) $(OBJECTS)
ifeq ($(DEPENDENCY_DB), true)
	@echo "[S_LIB_INFO] : $(shell pwd -P) : $(ARFLAGS) $(sort $(OBJECTS)) : $(STATIC_LIB)"
endif
	@echo "  Building static library $@."
	@$(MKINSTALLDIRS) $(OBJDIR)
	@rm -f $@
	$(AR) $(ARFLAGS) $@ $(sort $(OBJECTS))
	@echo "                        ******************************"
	@echo "                           $@"
	@echo "                        ******************************"

$(DYNAMIC_LIB): $(OBJECTS)
# It is workaround to avoid race-condition at parallel processing of same recipe
	$(eval myLib:=$(LIBNAME).so.$(SUFFIX_MAKE))
	$(eval myTmpDir:=$(OBJDIR)/signtmp-$(myLib)-$(shell echo $$PPID))
	$(eval myTarget:=$$(myTmpDir)/$(myLib))
ifeq ($(DEPENDENCY_DB), true)
	@echo "[D_LIB_INFO] : $(shell pwd -P) : $(sort $(OBJECTS)) $(LDFLAGS) $(ldFlagOverride) : $(DYNAMIC_LIB)"
endif
	@echo "  Building dynamic library $@."
	@mkdir -p $(myTmpDir)
	@$(MKINSTALLDIRS) $(OBJDIR) $(PGM_DIR) $(CFG_DIR) $(NOSTRIP_MAP_DIR)
ifeq ($(OBJ_HDR_SYM),hdr_sym)
	$(CC) $(sort $(OBJECTS)) -shared -Wl,-soname,$(DLIB_NAME) $(LDFLAGS) -o $(OBJDIR)/$(DLIB_1_0_1) $(ldFlagOverride)
else
	$(CC) $(LDFLAGS) -shared -Wl,-soname,$(DLIB_NAME) -o $(OBJDIR)/$(DLIB_1_0_1) $(ldFlagOverride) $(sort $(OBJECTS))
endif
	$(INSTALL) $(OBJDIR)/$(DLIB_1_0_1) $(myTarget)
ifneq ($(NM_BUILD),false)
	@$(NM) -anl $(OBJDIR)/$(DLIB_1_0_1) > $(OBJDIR)/$(DLIB_1_0_1).map
	@$(CP) -p -f $(OBJDIR)/$(DLIB_1_0_1).map $(OBJDIR)/$(myLib).map
	$(INSTALL) $(OBJDIR)/$(myLib).map $(NOSTRIP_MAP_DIR)
endif
	@$(CP) -p -f $(myTarget) $(OBJDIR)/$(myLib).nostrip
	$(INSTALL) $(OBJDIR)/$(myLib).nostrip $(NOSTRIP_MAP_DIR)
	cd $(OBJDIR) && ( ln -fs ./$(DLIB_1_0_1) $(DLIB_1); ln -fs ./$(DLIB_1_0_1) $(DLIB_NAME) )
ifeq (,$(findstring $(OBJDIR),OBJ/sim OBJ/tool))
ifeq ($(SANITIZE_STRIP_SKIP),true)
	@echo "Skip STRIP $(myTarget)"
else
	@$(STRIP) $(myTarget)
endif
endif
# To avoid duplicatd signing and mkfilever
ifdef TOOLCHAIN_GDB_PATH
	cd $(dir $(myTarget)) && $(PROJECT_UTIL_DIR)/mkqbinfohdr $(notdir $(myTarget)) $(SUFFIX_MAKE) $(DOCKER_IMAGE_NAME) $(TOOLCHAIN_GDB_PATH) $(TOOLCHAIN_SYSROOT_PATH)
endif
	$(PROJECT_UTIL_DIR)/mksigning $(myTarget) $(myTarget) $(CODE_SIGNING_KEY)
	$(PROJECT_UTIL_DIR)/mkfilever $(notdir $(myTarget)) $(Version) $(dir $(myTarget))
ifndef NON_INSTALL
	$(INSTALL) $(myTarget) $(PGM_DIR)/$(myLib)
ifneq ($(GZIP),)
	gzip -cn $(myTarget) > $(GZIP)
	$(INSTALL) $(GZIP) $(PGM_DIR)
endif
endif
ifneq ($(RFS_COPY),NONE)
ifeq ($(ADD_FW),on)
ifneq ($(RFS_ROOM_BASE),)
ifneq ($(RFS_ROOM_PATH),)
	@echo $(RFS_ROOM_PATH)
	@$(foreach cp_path,$(RFS_ROOM_PATH),echo " +++ RFS ROOM COPY $(INSTALL) $(myTarget) $(RFS_ROOM_BASE)/$(cp_path)";$(INSTALL) $(myTarget) $(RFS_ROOM_BASE)/$(cp_path);)
endif
endif
endif
endif
	@echo "                        ******************************"
	@echo "                           $@"
	@echo "                        ******************************"
	mv $(myTarget) $(OBJDIR)/$(myLib)
	rm -rf $(myTmpDir)

%.a:
	@echo "  Checking Static Library $@ at $$(pwd)"
	@$(MKINSTALLDIRS) $(@D)
	@if [ -f $(pathdir)/Makefile.arch_all ];then \
		export USER_SUFFIX=$(SUFFIX_MAKE);\
		cd $(pathdir) && flock -x Makefile.arch_all $(MAKE) -f Makefile.arch_all $(MAKE_OPTIONS) RFS_COPY=NONE;\
	else \
		cd $(pathdir) && flock -x Makefile.$(SUFFIX_MAKE) $(MAKE) -f Makefile.$(SUFFIX_MAKE) $(MAKE_OPTIONS) RFS_COPY=NONE;\
	fi

ifneq ($(NM_BUILD),false)
	@$(NM) -anl $@ > $@.map
endif
	@echo; echo "make[$(MAKELEVEL)]: Returning directory \`$(shell pwd)'"

endif
